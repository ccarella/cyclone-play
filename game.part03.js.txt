ion stormStutter() {
    if (state.stutterT > 0 || state.missLockT > 0) return;
    state.stutterT = STUTTER_S;
    state.missLockT = MISS_LOCK_S;
    state.pipFlash = 0.45;
    addPipPop("MISS", CX, CY - 48, PAL.enemy);
    addFx("stutter", { life: 0.22 });
    state.shake = Math.max(state.shake, 4.2);
    beep(70, 0.14, "sawtooth", 0.055);
    beep(110, 0.1, "square", 0.03);
  }

  function chargeFill() {
    return clamp(state.chargeT / CHARGE_MAX_S, 0, 1);
  }

  function chargeReady() {
    return state.chargeT >= CHARGE_MIN_S;
  }

  function chargingNow() {
    return input.charging && input.down && state.chargeT > 0;
  }

  function cancelCharge() {
    state.chargeT = 0;
    input.charging = false;
    input.wantAim = null;
  }

  function steerAim(targetA, dt) {
    const d = angNorm(targetA - state.aim);
    if (chargeReady()) {
      const max = CHARGE_AIM_RATE * dt;
      state.aim += clamp(d, -max, max);
    } else {
      state.aim = targetA;
    }
  }

  function fireTwister() {
    if (state.mode !== "play") {
      cancelCharge();
      return;
    }
    if (!chargeReady()) {
      cancelCharge();
      return;
    }
    const pipHit = pipOnTarget();
    cancelCharge();
    state.chargeCd = CHARGE_CD_S;
    state.chargeFlash = 0.38;
    state.shake = Math.max(state.shake, 7.2);
    addFx("twister", { a: state.aim, life: 0.48 });
    addPop("TWISTER", CX, CY - 40, PAL.gold);
    beep(90, 0.22, "sawtooth", 0.07);
    beep(220, 0.16, "triangle", 0.05);
    beep(740, 0.1, "square", 0.045);
    for (const e of state.enemies) {
      if (e.dead) continue;
      if (angDiff(e.a, state.aim) <= TWISTER_CONE) {
        hitEnemy(e);
      } else {
        e.r = clamp(e.r + TWISTER_KNOCK, DEATH_R + 4, RING_R - 2);
        e.flash = 0.7;
      }
    }
    if (state.boss && !state.boss.dead) {
      if (angDiff(state.boss.a, state.aim) <= TWISTER_CONE) hitBoss(3, 18);
      else state.boss.r = clamp(state.boss.r + TWISTER_KNOCK, DEATH_R + 4, RING_R - 2);
    }
    sliceShards(state.aim, TWISTER_CONE);
    for (const p of state.pickups) p.pulled = true;
    if (state.pickups.length) addPop("PULL", CX, CY - 22, PAL.gold);
    if (pipHit) perfectStop();
    else stormStutter();
  }

  function tickCharge(dt) {
    if (state.mode !== "play") return;
    if (!input.down || !input.charging || state.chargeCd > 0) return;
    const prev = state.chargeT;
    state.chargeT = Math.min(CHARGE_MAX_S, state.chargeT + dt);
    const fill = chargeFill();
    if (prev < CHARGE_MIN_S && state.chargeT >= CHARGE_MIN_S) {
      beep(620, 0.06, "square", 0.04);
    }
    const step = Math.floor(state.chargeT * 8);
    if (step !== Math.floor(prev * 8) && chargeReady()) {
      beep(240 + fill * 520, 0.04, "square", 0.025);
    }
    if (input.wantAim != null) steerAim(input.wantAim, dt);
    if (state.chargeT >= CHARGE_MAX_S) fireTwister();
  }

  function resetSwirl() {
    state.swirl = [];
    for (let i = 0; i < 18; i++) {
      state.swirl.push({
        a: rand(0, Math.PI * 2),
        r: rand(22, 42),
        s: rand(2.2, 4.4),
      });
    }
  }

  function isBossLevel(n) {
    return n >= BOSS_EVERY && n % BOSS_EVERY === 0;
  }

  function levelsToBoss(n) {
    if (isBossLevel(n)) return 0;
    return BOSS_EVERY - (n % BOSS_EVERY);
  }

  function iceCovers(a) {
    return !!(state.ice && angDiff(a, state.ice.a) <= state.ice.half);
  }

  function grantSteal() {
    if (state.steal === STEAL_ID) return;
    state.steal = STEAL_ID;
    addPop("STEAL", CX, CY - 52, PAL.ice, { scale: 3, life: 1.2 });
    addPop(STEAL_NAME, CX, CY - 28, PAL.cyan, { scale: 2, life: 1.4 });
    beep(520, 0.08, "square", 0.05);
    beep(780, 0.1, "triangle", 0.045);
    beep(1180, 0.16, "sine", 0.04);
  }

  function applyFreezeSpin() {
    state.ice = { a: state.aim, half: ICE_HALF, t: 0, life: ICE_LIFE };
    addFx("freeze", { a: state.aim, half: ICE_HALF, life: 0.48 });
    addPop("FREEZE", CX, CY - 40, PAL.ice);
    beep(1480, 0.07, "sine", 0.04);
    beep(1860, 0.11, "triangle", 0.035);
    sliceShards(state.aim, ICE_HALF);
  }

  function enterOrbit(b, delay) {
    b.phase = "orbit";
    b.phaseT = 0;
    b.telFill = 0;
    b.nextAtk = delay ?? 1.15;
    b.lungeSpd = LUNGE_SPD;
  }

  function startBossAtk(b) {
    b.phaseT = 0;
    b.telFill = 0;
    if (b.nextKind === "slam") {
      b.phase = "windup_slam";
      b.telA = angNorm(b.a + Math.PI * 0.72);
      b.telHalf = 0.88;
      b.nextKind = "lunge";
      addPop("SLAM", CX, CY - 36, PAL.gold);
      beep(180, 0.08, "sawtooth", 0.04);
      beep(240, 0.1, "square", 0.03);
    } else {
      b.phase = "windup_lunge";
      b.telA = b.a;
      b.telHalf = 0.5;
      b.nextKind = "slam";
      addPop("LUNGE", CX, CY - 36, PAL.enemy);
      beep(110, 0.1, "sawtooth", 0.045);
    }
  }

  function spawnBoss() {
    const a = angNorm(state.aim + Math.PI);
    state.boss = {
      hp: BOSS_HP,
      maxHp: BOSS_HP,
      a,
      r: RING_R - 6,
      phase: "orbit",
      phaseT: 0,
      nextAtk: 1.45,
      nextKind: "lunge",
      telA: a,
      telHalf: 0.5,
      telFill: 0,
      shards: [],
      flash: 0,
      wobble: 0,
      orbitDir: Math.random() < 0.5 ? -1 : 1,
      lungeSpd: LUNGE_SPD,
      dead: false,
      clearT: 0,
    };
  }
